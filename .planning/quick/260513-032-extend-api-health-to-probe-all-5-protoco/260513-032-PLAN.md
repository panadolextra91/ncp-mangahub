<canonical_truth>
- HTTP+WS share port `cfg.Port` (8080). WS probe = "is HTTP serving?" which we already are (self-check returns "ok"). Do NOT do a real WebSocket handshake — overkill for academic demo and adds gorilla/websocket client dep where not needed.
- TCP probe: `net.DialTimeout("tcp", "127.0.0.1:"+cfg.TCPPort, 500*time.Millisecond)` then immediate `conn.Close()`. Success = "ok", err = "error: <err.Error()>".
- UDP probe: `net.Dial("udp", "127.0.0.1:"+cfg.UDPPort)` — UDP is connectionless; Dial just resolves the address and binds a local socket. If `Dial` returns no error, write a single "PING\n" byte and close. We do NOT wait for a response. Liveness = "did the local kernel let us send a packet to that port". This is pragmatic and matches the bonus rubric.
- gRPC probe: use `grpc.NewClient("127.0.0.1:"+cfg.GRPCPort, grpc.WithTransportCredentials(insecure.NewCredentials()))` then a short `WaitForStateChange`/`Connect` with a 500ms `context.WithTimeout`. If the conn reaches `Ready` or `Connecting` state, "ok". On timeout, "error: timeout". Close conn after. (Use `grpc.NewClient` since `grpc.Dial` + `WithBlock` is deprecated in newer grpc-go; check go.sum, fall back to `grpc.Dial` with `WithBlock` + `WithTimeout` if `NewClient` not available.)
- DB probe: keep existing `h.db.Ping()`.
- HTTP self: always "ok" (we ARE responding).
- uptime: `time.Since(startedAt).Seconds()` rounded to 2 decimal places.
- All 6 probes run as goroutines with `sync.WaitGroup`. Results written to a `map[string]string` guarded by `sync.Mutex` (small map, no need for sync.Map).
- Total endpoint latency: ~500ms worst case (slowest probe dominates).

**Breaking change to response body** (document in summary): the `status` field semantic changes from `"MangaHub is alive"` (greeting string) to `"ok"|"degraded"` (state enum). The `db` field changes from top-level string `"OK"|"ERROR"` to nested `checks.db: "ok"|"error: ..."`. `bus.dropped_events` is preserved. `timestamp` is preserved.
</canonical_truth>

       ```go
       func (h *HealthHandler) Check(w http.ResponseWriter, r *http.Request) {
           log.Printf("🩺 Health check from %s at %s", r.RemoteAddr, time.Now().UTC().Format(time.RFC3339))

           const probeTimeout = 500 * time.Millisecond
           results := map[string]string{}
           var mu sync.Mutex
           var wg sync.WaitGroup

           set := func(name, val string) { mu.Lock(); results[name] = val; mu.Unlock() }

           // 1. HTTP self-check
           wg.Add(1)
           go func() { defer wg.Done(); set("http", "ok") }()

           // 2. TCP probe
           wg.Add(1)
           go func() {
               defer wg.Done()
               conn, err := net.DialTimeout("tcp", "127.0.0.1:"+h.tcpPort, probeTimeout)
               if err != nil { set("tcp", "error: "+err.Error()); return }
               conn.Close(); set("tcp", "ok")
           }()

           // 3. UDP probe — best-effort send (UDP is connectionless)
           wg.Add(1)
           go func() {
               defer wg.Done()
               conn, err := net.Dial("udp", "127.0.0.1:"+h.udpPort)
               if err != nil { set("udp", "error: "+err.Error()); return }
               defer conn.Close()
               conn.SetWriteDeadline(time.Now().Add(probeTimeout))
               if _, err := conn.Write([]byte("PING\n")); err != nil {
                   set("udp", "error: "+err.Error()); return
               }
               set("udp", "ok")
           }()

           // 4. WS self-check (HTTP server up == WS upgrade endpoint reachable)
           wg.Add(1)
           go func() { defer wg.Done(); set("ws", "ok") }()

           // 5. gRPC probe — Dial+WaitForReady within 500ms
           wg.Add(1)
           go func() {
               defer wg.Done()
               ctx, cancel := context.WithTimeout(context.Background(), probeTimeout)
               defer cancel()
               // Use grpc.NewClient (preferred) or grpc.DialContext if NewClient unavailable
               conn, err := grpc.NewClient("127.0.0.1:"+h.grpcPort, grpc.WithTransportCredentials(insecure.NewCredentials()))
               if err != nil { set("grpc", "error: "+err.Error()); return }
               defer conn.Close()
               conn.Connect()
               // Poll state until Ready or context expires
               for {
                   s := conn.GetState()
                   if s.String() == "READY" { set("grpc", "ok"); return }
                   if !conn.WaitForStateChange(ctx, s) {
                       set("grpc", "error: timeout (last state "+s.String()+")"); return
                   }
               }
           }()

           // 6. DB probe
           wg.Add(1)
           go func() {
               defer wg.Done()
               if err := h.db.Ping(); err != nil { set("db", "error: "+err.Error()); return }
               set("db", "ok")
           }()

           wg.Wait()

           overall := "ok"
           for _, v := range results {
               if v != "ok" { overall = "degraded"; break }
           }

           uptime := math.Round(time.Since(h.startedAt).Seconds()*100) / 100

           log.Printf("🩺 Health check result: status=%s http=%s tcp=%s udp=%s ws=%s grpc=%s db=%s",
               overall, results["http"], results["tcp"], results["udp"], results["ws"], results["grpc"], results["db"])

           w.Header().Set("Content-Type", "application/json")
           if overall == "degraded" {
               w.WriteHeader(http.StatusServiceUnavailable)
           } else {
               w.WriteHeader(http.StatusOK)
           }
           json.NewEncoder(w).Encode(map[string]interface{}{
               "status":         overall,
               "checks":         results,
               "bus":            map[string]interface{}{"dropped_events": h.bus.DroppedCount()},
               "uptime_seconds": uptime,
               "timestamp":      time.Now().UTC().Format(time.RFC3339),
           })

           _ = fmt.Sprintf // keep fmt import alive if not used elsewhere; remove if executor doesn't need fmt
       }
       ```
